function saveSnapshot(incidents, orders, audit = true) {
  const insertI = db.prepare('INSERT OR REPLACE INTO incidents(id,data,updated_at) VALUES(?,?,?)');
  const insertW = db.prepare('INSERT OR REPLACE INTO work_orders(id,data,updated_at) VALUES(?,?,?)');
  const now = new Date().toISOString();
  db.exec('BEGIN IMMEDIATE');
  try {
    db.exec('DELETE FROM incidents; DELETE FROM work_orders;');
    for (const x of incidents) insertI.run(x.id, JSON.stringify(x), now);
    for (const x of orders) insertW.run(x.id, JSON.stringify(x), now);
    db.prepare("INSERT OR REPLACE INTO app_meta(key,value) VALUES('last_updated',?)").run(now);
    if (audit) db.prepare('INSERT INTO audit_log VALUES(?,?,?,?,?,?)').run(randomUUID(),'snapshot_saved','workspace','all',now,JSON.stringify({incidents:incidents.length,workOrders:orders.length}));
    db.exec('COMMIT');
  } catch (error) { db.exec('ROLLBACK'); throw error; }
}

// ---- Safe saving of the workspace -------------------------------------------------
// Several screens change incidents and work orders. The Operations desk sends its whole copy back on every edit,
// so a stale copy could erase newer changes. Each snapshot has a version; a stale save is merged (3-way) instead of overwriting.
db.exec('CREATE TABLE IF NOT EXISTS snapshot_history (version TEXT PRIMARY KEY, data TEXT NOT NULL, created_at TEXT NOT NULL)');
function canon(v) { if (v === undefined) return 'u'; if (v === null || typeof v !== 'object') return JSON.stringify(v); if (Array.isArray(v)) return '[' + v.map(canon).join(',') + ']'; return '{' + Object.keys(v).sort().map(k => JSON.stringify(k) + ':' + canon(v[k])).join(',') + '}'; }
function currentSnapshot() { return { incidents: readRows('incidents'), workOrders: readRows('work_orders') }; }
function snapshotVersion(s) { const byId = (a, b) => (a.id < b.id ? -1 : 1); return createHash('sha1').update(canon({ i: [...s.incidents].sort(byId), w: [...s.workOrders].sort(byId) })).digest('hex'); }
function rememberSnapshot(s) {
  const v = snapshotVersion(s);
  try { db.prepare('INSERT OR IGNORE INTO snapshot_history VALUES(?,?,?)').run(v, JSON.stringify(s), new Date().toISOString()); db.prepare('DELETE FROM snapshot_history WHERE version NOT IN (SELECT version FROM snapshot_history ORDER BY created_at DESC LIMIT 40)').run(); } catch (_) {}
  return v;
}
function mergeSnapshots(base, client, server) {
  const out = {};
  for (const key of ['incidents', 'workOrders']) {
    const m = list => new Map(list.map(x => [x.id, x])), b = m(base[key]), c = m(client[key]), s = m(server[key]), res = [];
    for (const id of new Set([...b.keys(), ...c.keys(), ...s.keys()])) {
      const bx = b.get(id), cx = c.get(id), sx = s.get(id), clientChanged = canon(bx) !== canon(cx), serverChanged = canon(bx) !== canon(sx);
      const pick = !clientChanged ? sx : (!serverChanged ? cx : cx);
      if (pick) res.push(pick);
    }
    out[key] = res;
  }
  return out;
}

async function runYolo(imagePath) {
  if (!fs.existsSync(MODEL_PATH)) return { status: 'model_not_configured', detections: [] };
  try {
    const script = path.join(ROOT, 'ai', 'infer.py');
    const args = PYTHON === 'py' ? ['-3', script, '--weights', MODEL_PATH, '--image', imagePath] : [script, '--weights', MODEL_PATH, '--image', imagePath];
    const { stdout } = await execFileAsync(PYTHON, args, { cwd: ROOT, timeout: 90000, maxBuffer: 2 * 1024 * 1024, windowsHide: true });
    const result = JSON.parse(stdout.trim());
    return result.ok ? result : { status: result.status || 'inference_error', message: result.message || 'YOLO inference failed.', detections: [] };
  } catch (error) {
    return { status: error.killed ? 'inference_timeout' : 'inference_unavailable', message: String(error.message || error).slice(0, 300), detections: [] };
  }
}
function classifyYolo(detections) {
  const d = detections?.[0]; if (!d) return null;
  const n = String(d.className).toLowerCase().replace(/[_-]+/g, ' ');
  let type = 'Other';
  if (/^d40$/.test(n) || /pothole|hole/.test(n)) type = 'Pothole';
  else if (/^(d00|d01|d10|d11|d20)$/.test(n) || /crack|alligator|rut|surface defect/.test(n)) type = 'Road crack';
  else if (/^(road edge breakage|asphalt ravelling|failed repair patch|surface bleeding)$/.test(n)) type = 'Road crack';
  else if (/sign/.test(n)) type = 'Damaged sign';
  else if (/light|lamp/.test(n)) type = 'Broken streetlight';
  else if (/drain|utility cover/.test(n)) type = 'Blocked drain';
  return { type, confidence: d.confidence, className: d.className };
}

const tokenHash = token => createHash('sha256').update(String(token)).digest('hex');
function dropSession(token) { sessions.delete(token); try { db.prepare('DELETE FROM sessions WHERE token_hash=?').run(tokenHash(token)); } catch (_) {} }
function normalizeEmail(value) { return String(value || '').trim().toLowerCase(); }
function passwordRecord(password) {
  const salt = randomBytes(16).toString('hex');
  const hash = scryptSync(String(password), salt, 64).toString('hex');
  return { salt, hash };
}
function verifyPassword(password, salt, hash) {
  try {
    const actual = scryptSync(String(password), salt, 64);
    const expected = Buffer.from(hash, 'hex');
    return actual.length === expected.length && timingSafeEqual(actual, expected);
  } catch (_) { return false; }
}
function sessionUser(req) {
  const raw = String(req.headers.cookie || '').split(';').map(x => x.trim()).find(x => x.startsWith('ss_session='));
  if (!raw) return null;
  const token = decodeURIComponent(raw.slice('ss_session='.length));
  let session = sessions.get(token);
  if (!session) { const r = db.prepare('SELECT user_id AS userId, expires_at AS expiresAt FROM sessions WHERE token_hash=?').get(tokenHash(token)); if (r) { session = { userId: r.userId, expiresAt: r.expiresAt }; sessions.set(token, session); } }
  if (!session || session.expiresAt < Date.now()) { dropSession(token); return null; }
  const user = db.prepare('SELECT id,email,display_name AS displayName,role,active,created_at AS createdAt FROM users WHERE id=?').get(session.userId);
  if (!user || !user.active) { dropSession(token); return null; }
  return user;
}
function setSession(res, userId) {
  const token = randomBytes(32).toString('hex');
  const expiresAt = Date.now() + SESSION_TTL; sessions.set(token, { userId, expiresAt });
  try { db.prepare('DELETE FROM sessions WHERE expires_at < ?').run(Date.now()); db.prepare('INSERT OR REPLACE INTO sessions VALUES(?,?,?)').run(tokenHash(token), userId, expiresAt); } catch (_) {}
  res.setHeader('Set-Cookie', `ss_session=${token}; HttpOnly; SameSite=Lax; Path=/; Max-Age=${Math.floor(SESSION_TTL/1000)}${COOKIE_SECURE}`);
}
function clearSession(req, res) {
  const raw = String(req.headers.cookie || '').split(';').map(x => x.trim()).find(x => x.startsWith('ss_session='));
  if (raw) dropSession(decodeURIComponent(raw.slice('ss_session='.length)));
  res.setHeader('Set-Cookie', 'ss_session=; HttpOnly; SameSite=Lax; Path=/; Max-Age=0' + COOKIE_SECURE);
}

          if (!buffer.length || buffer.length>8*1024*1024) return json(res,400,{error:'Photo must be 8 MB or smaller.'});
          // If the citizen did not enter coordinates, use GPS embedded in the original image.
          if (lat === null && lng === null) {

            if (embeddedGps) {
              lat = embeddedGps.lat;
              lng = embeddedGps.lng;
            }
          }
          imageName=randomUUID()+ext;
          fs.writeFileSync(path.join(EVIDENCE_DIR,imageName),buffer,{flag:'wx'});
        }
        if ((lat === null) !== (lng === null) || (lat !== null && (!Number.isFinite(lat) || !Number.isFinite(lng) || lat < -90 || lat > 90 || lng < -180 || lng > 180))) return json(res,400,{error:'Enter valid latitude and longitude coordinates.'});
        let aiAnalysis = {status:'not_run',model:null,detections:[],prediction:null,message:imageName ? null : 'No photo was uploaded.'};
        if (imageName) {
          const analysis = await runYolo(path.join(EVIDENCE_DIR,imageName));
          const prediction = analysis.status === 'inference_complete' ? classifyYolo(analysis.detections) : null;
          aiAnalysis = {status:analysis.status,model:analysis.model||null,detections:analysis.detections||[],prediction,message:analysis.message||null,scene:analysis.scene||null};
          if (prediction) audit('yolo_inference_completed','complaint',id,{className:prediction.className,confidence:prediction.confidence});
        }
        const complaint={id,category,location,lat,lng,description,status:'Submitted',priority:'Awaiting review',image:imageName?`/api/complaint-image/${imageName}`:null,reporter:{id:currentUser.id,email:currentUser.email,name:currentUser.displayName},createdAt:now,updatedAt:now,adminNote:'',aiAnalysis};
        db.prepare('INSERT INTO complaints(id,user_id,data,created_at,updated_at) VALUES(?,?,?,?,?)').run(id,currentUser.id,JSON.stringify(complaint),now,now);
        audit('citizen_complaint_submitted','complaint',id,{category,userId:currentUser.id});
        insights.autoApproveComplaint(id); const fresh=db.prepare('SELECT data FROM complaints WHERE id=?').get(id);
        return json(res,201,{item:fresh?JSON.parse(fresh.data):complaint});
      }
      if (req.method === 'GET' && p === '/api/complaints') {
        if (!currentUser) return json(res,401,{error:'Please sign in.'});
        insights.syncComplaints();
        const rows=currentUser.role==='admin'
          ? db.prepare('SELECT data FROM complaints ORDER BY created_at DESC').all()
          : db.prepare('SELECT data FROM complaints WHERE user_id=? ORDER BY created_at DESC').all(currentUser.id);
        return json(res,200,{items:rows.map(safeComplaint)});
      }
      if (req.method === 'GET' && p === '/api/admin/complaints') {
        if (!currentUser || currentUser.role!=='admin') return json(res,403,{error:'Administrator access required.'});
        insights.syncComplaints();
        return json(res,200,{items:db.prepare('SELECT data FROM complaints ORDER BY created_at DESC').all().map(safeComplaint)});
      }
      const complaintMatch=p.match(/^\/api\/admin\/complaints\/([^/]+)$/);
      if (complaintMatch && req.method==='PATCH') {
        if (!currentUser || currentUser.role!=='admin') return json(res,403,{error:'Administrator access required.'});
        const row=db.prepare('SELECT data FROM complaints WHERE id=?').get(complaintMatch[1]);
        if (!row) return json(res,404,{error:'Complaint not found.'});
        const item=JSON.parse(row.data), data=await bodyJson(req);
        const statuses=['Submitted','Under review','Assigned','In progress','Resolved','Rejected'];
        if (data.status && !statuses.includes(data.status)) return json(res,400,{error:'Invalid complaint status.'});
        if (data.status) { item.status=data.status; insights.onComplaintStatus(item,data.status); }
        if (data.adminNote !== undefined) item.adminNote=String(data.adminNote).slice(0,1000);
        item.updatedAt=new Date().toISOString();
        db.prepare('UPDATE complaints SET data=?,updated_at=? WHERE id=?').run(JSON.stringify(item),item.updatedAt,item.id);
        audit('complaint_updated','complaint',item.id,{status:item.status,by:currentUser.id});
        return json(res,200,{item});
      }
      const delMatch=p.match(/^\/api\/complaints\/([^/]+)$/);
      if (delMatch && req.method==='DELETE') {
        if (!currentUser) return json(res,401,{error:'Sign in required.'});
        const row=db.prepare('SELECT user_id,data FROM complaints WHERE id=?').get(delMatch[1]);
        if (!row) return json(res,404,{error:'Report not found.'});
        const isAdmin=currentUser.role==='admin', item=JSON.parse(row.data);
        if (!isAdmin && row.user_id!==currentUser.id) return json(res,404,{error:'Report not found.'});
        if (!isAdmin && !['Submitted','Under review','Rejected'].includes(item.status)) return json(res,409,{error:'This report is already being handled. Ask an administrator to remove it.'});
        const img=path.basename(String(item.imagePath||item.image||'')); if (img && /^[a-f0-9-]+\.(jpg|png|webp)$/i.test(img)) { try { fs.rmSync(path.join(EVIDENCE_DIR,img),{force:true}); } catch(_){} }
        db.prepare('DELETE FROM complaints WHERE id=?').run(item.id);
        audit('complaint_deleted','complaint',item.id,{by:currentUser.id,admin:isAdmin});
        return json(res,200,{deleted:item.id});
      }
      const imageMatch=p.match(/^\/api\/complaint-image\/([a-f0-9-]+\.(?:jpg|png|webp))$/i);
      if (imageMatch && req.method==='GET') {
        if (!currentUser) return json(res,401,{error:'Sign in required.'});
        const imagePath=path.join(EVIDENCE_DIR,path.basename(imageMatch[1]));
        const row=db.prepare('SELECT user_id,data FROM complaints WHERE data LIKE ?').get(`%${imageMatch[1]}%`);
        if (!row || (currentUser.role!=='admin' && row.user_id!==currentUser.id) || !fs.existsSync(imagePath)) return json(res,404,{error:'Image not found.'});
        const ext=path.extname(imagePath), type=({'.jpg':'image/jpeg','.png':'image/png','.webp':'image/webp'})[ext];
        res.writeHead(200,{'Content-Type':type,'Cache-Control':'private, max-age=300','X-Content-Type-Options':'nosniff'});
        return fs.createReadStream(imagePath).pipe(res);
      }

      if (await insights.pub(req,res,p,url)) return;

      if (p !== '/api/health' && p.startsWith('/api/')) {
        if (!currentUser) return json(res,401,{error:'Sign in required.'});
        if (currentUser.role !== 'admin') return json(res,403,{error:'Administrator access required.'});
      }
      if (await insights.admin(req,res,p,url)) return;
      if (req.method === 'GET' && p === '/api/health') return json(res,200,{ok:true,build:'20',service:'StreetSense API',database:'SQLite',mode:'local-city operations',evidenceStorage:true,aiModel:fs.existsSync(MODEL_PATH)?'yolo-weights-present':'not-configured',time:new Date().toISOString()});
      if (req.method === 'GET' && p === '/api/snapshot') { const snap=currentSnapshot(); return json(res,200,{version:rememberSnapshot(snap),incidents:snap.incidents,workOrders:snap.workOrders,lastUpdated:db.prepare("SELECT value FROM app_meta WHERE key='last_updated'").get()?.value || null}); }
      if (req.method === 'PUT' && p === '/api/snapshot') {
        const data=await bodyJson(req); const incidents=data.incidents, orders=data.workOrders;
        if (!Array.isArray(incidents)||!Array.isArray(orders)||incidents.length>5000||orders.length>5000||!incidents.every(validIncident)||!orders.every(validOrder)) return json(res,400,{error:'Invalid workspace data. Check incident and work-order fields.'});
        const cur=currentSnapshot(), curVersion=snapshotVersion(cur), baseVersion=typeof data.baseVersion==='string'?data.baseVersion:null;
        if (!baseVersion || baseVersion===curVersion) { saveSnapshot(incidents,orders,true); const v=rememberSnapshot(currentSnapshot()); return json(res,200,{ok:true,version:v,merged:false,savedAt:new Date().toISOString(),incidents:incidents.length,workOrders:orders.length}); }
        const baseRow=db.prepare('SELECT data FROM snapshot_history WHERE version=?').get(baseVersion);
        if (!baseRow) return json(res,200,{ok:false,conflict:true,version:rememberSnapshot(cur),incidents:cur.incidents,workOrders:cur.workOrders});
        const merged=mergeSnapshots(JSON.parse(baseRow.data),{incidents,workOrders:orders},cur);
        saveSnapshot(merged.incidents,merged.workOrders,true); db.prepare('INSERT INTO audit_log VALUES(?,?,?,?,?,?)').run(randomUUID(),'snapshot_merged','workspace','all',new Date().toISOString(),JSON.stringify({incidents:merged.incidents.length,workOrders:merged.workOrders.length}));
        return json(res,200,{ok:true,merged:true,version:rememberSnapshot(currentSnapshot()),incidents:merged.incidents,workOrders:merged.workOrders});
      }
      if (req.method === 'POST' && p === '/api/evidence') {
        const data = await bodyJson(req);
        const allowed = new Set(['image/jpeg','image/png','image/webp']);
        if (!data || typeof data.filename !== 'string' || data.filename.length > 180 || !allowed.has(data.mimeType) || typeof data.base64 !== 'string') return json(res,400,{error:'Upload a JPG, PNG, or WebP image.'});
        const buffer = Buffer.from(data.base64, 'base64');
        if (!buffer.length || buffer.length > 8*1024*1024 || buffer.toString('base64').replace(/=+$/,'') !== data.base64.replace(/=+$/,'')) return json(res,400,{error:'Evidence must be a valid image up to 8 MB.'});
        const id = randomUUID(); const ext = ({'image/jpeg':'.jpg','image/png':'.png','image/webp':'.webp'})[data.mimeType]; const storedName = `${id}${ext}`;
        fs.writeFileSync(path.join(EVIDENCE_DIR, storedName), buffer, {flag:'wx'});
        const lat = Number.isFinite(Number(data.lat)) && data.lat !== null ? Number(data.lat) : null;
        const lng = Number.isFinite(Number(data.lng)) && data.lng !== null ? Number(data.lng) : null;
        if ((lat === null) !== (lng === null) || (lat !== null && (lat < -90 || lat > 90 || lng < -180 || lng > 180))) { fs.unlinkSync(path.join(EVIDENCE_DIR, storedName)); return json(res,400,{error:'Invalid GPS coordinates.'}); }
        const now = new Date().toISOString();

        const analysis = await runYolo(path.join(EVIDENCE_DIR, storedName));
        const prediction = analysis.status === 'inference_complete' ? classifyYolo(analysis.detections) : null;
        if (prediction) audit('yolo_inference_completed','evidence',id,{type:prediction.type,confidence:prediction.confidence});
        return json(res,201,{id,filename:path.basename(data.filename),lat,lng,reviewStatus:'pending_human_review',createdAt:now,modelStatus:analysis.status,analysis:{status:analysis.status,model:analysis.model||null,detections:analysis.detections||[],prediction,message:analysis.message||null}});
      }
      if (req.method === 'GET' && p === '/api/evidence') return json(res,200,{items:db.prepare('SELECT id,filename,mime_type AS mimeType,lat,lng,review_status AS reviewStatus,created_at AS createdAt FROM evidence ORDER BY created_at DESC LIMIT 100').all()});

    const realExt = path.extname(realParts[realParts.length - 1] || '').toLowerCase();
    if (!file.startsWith(ROOT + path.sep) || hiddenNow || outside || !publicExt.has(path.extname(file).toLowerCase()) || !publicExt.has(realExt) || !fs.existsSync(file) || !fs.statSync(file).isFile()) { res.writeHead(404); return res.end('Not Found'); }
    res.writeHead(200,{'Content-Type':mime(file),'Cache-Control':['.html','.js','.css','.svg'].includes(path.extname(file))?'no-cache':'public, max-age=3600'});
    if(req.method==='HEAD') return res.end(); fs.createReadStream(file).pipe(res);
  } catch (err) { console.error(err); json(res,400,{error:err instanceof SyntaxError?'Malformed JSON':err.message||'Request failed'}); }
});
